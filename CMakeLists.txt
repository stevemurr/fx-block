cmake_minimum_required(VERSION 3.22)
project(FxBlock VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FXBLOCK_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(FXBLOCK_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -DFXBLOCK_SANITIZED)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

add_library(fxblock_core STATIC src/Filter.cpp src/Drive.cpp src/Modulation.cpp src/Delay.cpp src/Reverb.cpp
  src/Width.cpp src/Chain.cpp)
target_include_directories(fxblock_core PUBLIC src)
set_target_properties(fxblock_core PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

add_library(FxBlock MODULE src/Plugin.cpp)
target_include_directories(FxBlock PRIVATE third_party/clap/include)
set_target_properties(FxBlock PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_library(fxblock_editor STATIC src/Editor.mm src/ui/Widgets.mm src/ui/Graphs.mm)
  target_compile_options(fxblock_editor PRIVATE -fobjc-arc)
  target_include_directories(fxblock_editor PUBLIC src)
  target_link_libraries(fxblock_editor PUBLIC fxblock_core "-framework AppKit" "-framework QuartzCore")
  set_target_properties(fxblock_editor PROPERTIES CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden)
  target_link_libraries(FxBlock PRIVATE fxblock_editor)
  set_target_properties(FxBlock PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION clap
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
  add_custom_command(TARGET FxBlock POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:FxBlock>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md"
      "$<TARGET_BUNDLE_DIR:FxBlock>/Contents/Resources/THIRD_PARTY_NOTICES.md"
    COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:FxBlock>"
    COMMENT "Ad-hoc signing the local CLAP bundle")
  install(TARGETS FxBlock LIBRARY DESTINATION . BUNDLE DESTINATION .)
else()
  target_link_libraries(FxBlock PRIVATE fxblock_core)
  set_target_properties(FxBlock PROPERTIES SUFFIX .clap)
  install(TARGETS FxBlock LIBRARY DESTINATION . RUNTIME DESTINATION .)
  install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(name Parameters Blocks Chain)
    add_executable(${name}_tests tests/${name}Tests.cpp)
    target_link_libraries(${name}_tests PRIVATE fxblock_core)
    add_test(NAME ${name} COMMAND ${name}_tests)
  endforeach()
  add_executable(plugin_tests tests/PluginTests.cpp src/Plugin.cpp)
  target_include_directories(plugin_tests PRIVATE third_party/clap/include)
  if(APPLE)
    target_link_libraries(plugin_tests PRIVATE fxblock_editor)
  else()
    target_link_libraries(plugin_tests PRIVATE fxblock_core)
  endif()
  add_test(NAME plugin COMMAND plugin_tests)
endif()

add_executable(render_demo tools/RenderDemo.cpp)
target_link_libraries(render_demo PRIVATE fxblock_core)
if(APPLE)
  add_executable(editor_snapshot tools/EditorSnapshot.mm)
  target_compile_options(editor_snapshot PRIVATE -fobjc-arc)
  target_link_libraries(editor_snapshot PRIVATE fxblock_editor)
endif()
